const SHEET_ID = '1-LULtf4-33rjaurLBIdWF2XsYYq13nLCVFnrjN8f_X4';

function doPost(e) {
  const data = JSON.parse(e.postData.contents);

  // El frontend manda varios tipos de payload a esta MISMA url (APPS_URL y
  // AJUSTE_URL apuntan al mismo /exec). Apps Script solo reconoce doPost
  // como punto de entrada -- doPostAjuste()/doPostUbicacion() de abajo
  // NUNCA se ejecutan solas, así que aquí distinguimos y enrutamos a mano.
  if (data.accion === 'ubicacion') {
    return doPostUbicacion(data);
  }
  if (data.accion === 'agregar_ubicacion_catalogo') {
    return doPostAgregarUbicacionCatalogo(data);
  }
  if (data.accion === 'eliminar_ubicacion') {
    return doPostEliminarUbicacion(data);
  }
  if (data.accion === 'pt_manual') {
    return doPostPtManual(data);
  }
  if (data.accion === 'eliminar_pt_manual') {
    return doPostEliminarPtManual(data);
  }
  if (data.accion === 'excluir_atrasado') {
    return doPostExcluirAtrasado(data);
  }
  if (data.accion === 'quitar_exclusion_atrasado') {
    return doPostQuitarExclusionAtrasado(data);
  }
  if (data.accion === 'agregar_lote') {
    return doPostAgregarLote(data);
  }
  if (data.accion === 'devolucion') {
    return doPostDevolucion(data);
  }
  if (data.accion === 'eliminar_devolucion') {
    return doPostEliminarDevolucion(data);
  }
  if (data.accion === 'plan_embarque') {
    return doPostPlanEmbarque(data);
  }
  if (data.fechaAjuste !== undefined) {
    return doPostAjuste(e, data);
  }
  return doPostConteo(data);
}

function doPostUbicacion(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);

  let ubic = ss.getSheetByName('Ubicaciones');
  if (!ubic) {
    ubic = ss.insertSheet('Ubicaciones');
    ubic.appendRow(['Id', 'Clave', 'Lote', 'Ubicacion', 'Registrado', 'Fecha', 'Tipo', 'Motivo']);
  }

  // Un lote puede tener varias ubicaciones a la vez (ej. una tarima
  // partida entre dos racks), así que cada guardado agrega una fila NUEVA
  // con su propio Id -- ya no se sobreescribe ninguna existente. Para
  // borrar una ubicación en particular se usa doPostEliminarUbicacion()
  // con ese Id.
  //
  // Tipo/Motivo (columnas 7-8) son para ubicaciones BLOQUEADAS -- casillas
  // del catálogo que están ocupadas por otro departamento (no por tu
  // inventario) y que se marcan a mano desde la pestaña "Ubicaciones" para
  // que no salgan como disponibles. En ese caso Clave/Lote vienen vacíos y
  // Tipo='Bloqueada' con el motivo. Un guardado normal (desde Inventario)
  // no manda data.tipo, así que Tipo/Motivo quedan en blanco como siempre.
  const id = Utilities.getUuid();
  const fila = ubic.getLastRow() + 1;
  const rango = ubic.getRange(fila, 1, 1, 8);
  // Fuerza texto plano ANTES de escribir: sin esto, Sheets interpreta
  // ubicaciones como "07E03" como notación científica y las guarda como
  // el número 7000 (se ve "7.00E+03" en vez del texto que escribiste).
  rango.setNumberFormat('@');
  rango.setValues([[id, data.clave || '', data.lote || '', data.ubicacion || '', data.registrado || '', data.fecha || '', data.tipo || '', data.motivo || '']]);

  return ContentService
    .createTextOutput(JSON.stringify({status:'OK', id:id}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPostEliminarUbicacion(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const ubic = ss.getSheetByName('Ubicaciones');
  if (ubic) {
    const vals = ubic.getDataRange().getValues();
    for (let i = 1; i < vals.length; i++) {
      if (String(vals[i][0]).trim() === String(data.id).trim()) {
        ubic.deleteRow(i + 1);
        break;
      }
    }
  }
  return ContentService
    .createTextOutput(JSON.stringify({status:'OK'}))
    .setMimeType(ContentService.MimeType.JSON);
}

// Crea una fila NUEVA en Ciclicos para un lote que nunca se sincronizó
// desde SAI (demasiado viejo) pero que existe físicamente y se confirmó en
// SAI. Hereda ABC/Semana/SemanaFecha de otra fila de la misma clave si ya
// está clasificada, igual que hace actualizarCiclicos() con lotes nuevos de
// una clave existente. No captura el conteo físico aquí -- el lote queda
// como cualquier otro renglón "Activo" sin contar, y se cuenta con el mismo
// input +✓ de siempre (con su regla de causa si la diferencia pasa del 3%
// de la existencia SAI de la clave).
function doPostAgregarLote(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const cic = ss.getSheetByName('Ciclicos');
  if (!cic) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'No existe la pestaña Ciclicos.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const clave = String(data.clave || '').trim();
  const lote = String(data.lote || '').trim();
  if (!clave || !lote) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'Clave y Lote son obligatorios.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const vals = cic.getDataRange().getValues();

  // No duplicar: si esa clave+lote ya existe, no crear otra fila.
  for (let i = 1; i < vals.length; i++) {
    if (String(vals[i][1]).trim() === clave && String(vals[i][3]).trim() === lote) {
      return ContentService
        .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'Ese lote ya existe en Cíclicos (fila ' + (i + 1) + ').' }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // Heredar clasificación de otra fila ya clasificada de la misma clave.
  let abc = '', semana = '', semanaFecha = '';
  for (let i = 1; i < vals.length; i++) {
    if (String(vals[i][1]).trim() === clave && vals[i][11]) {
      abc = vals[i][11];
      semana = vals[i][12];
      semanaFecha = vals[i][13];
      break;
    }
  }

  const clase = String(data.clase || 'EYE').trim();
  const descripcion = String(data.descripcion || '').trim();
  const existenciaSai = data.existenciaSai !== undefined && data.existenciaSai !== '' ? data.existenciaSai : '';
  const unidad = String(data.unidad || '').trim();
  const almacen = String(data.almacen || '').trim();

  cic.appendRow([clase, clave, descripcion, lote, existenciaSai, unidad, '', almacen, '', '', '', abc, semana, semanaFecha, 'Activo']);

  return ContentService
    .createTextOutput(JSON.stringify({ status: 'OK', abc: abc, semana: semana, semanaFecha: semanaFecha }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== PT agregado a mano (producción que aún no aparece en SAI) =====
// Se escribe directo en la hoja "InventarioPT" -- la MISMA que llena el otro
// script (actualizar inventario.txt) al sincronizar los reportes de Drive.
// Cada fila manual queda marcada con Origen='MANUAL' e IdManual único. El
// otro script, cuando corre el siguiente reporte real de ese almacén,
// reescribe toda esa sección de la hoja y estas filas manuales se pierden
// solas junto con las demás -- no hace falta borrarlas a mano.
function doPostPtManual(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let inv = ss.getSheetByName('InventarioPT');
  if (!inv) {
    inv = ss.insertSheet('InventarioPT');
    inv.appendRow(['Clase','Clave','Descripción','Lote','Existencia','Unidad','Fecha Caducidad','Almacén','Origen','IdManual']);
  }
  // Por si la hoja ya existía sin las columnas Origen/IdManual (la crea el
  // otro script al sincronizar), las agrega solas.
  const anchoHoy = Math.max(inv.getLastColumn(), 8);
  const encabezado = inv.getRange(1, 1, 1, anchoHoy).getValues()[0];
  if (String(encabezado[8] || '').trim() !== 'Origen') inv.getRange(1, 9).setValue('Origen');
  if (String(encabezado[9] || '').trim() !== 'IdManual') inv.getRange(1, 10).setValue('IdManual');

  const id = Utilities.getUuid();
  const fila = inv.getLastRow() + 1;
  const rango = inv.getRange(fila, 1, 1, 10);
  // Texto plano para que Clave/Lote no se conviertan en notación científica
  // (mismo problema que ya se corrigió en Ubicaciones).
  rango.setNumberFormat('@');
  rango.setValues([[
    data.clase || 'PT',
    data.clave || '',
    data.desc || '',
    data.lote || '',
    data.cantidad || 0,
    data.unidad || 'pzas',
    data.caducidad || '',
    data.almacen || '',
    'MANUAL',
    id
  ]]);

  return ContentService
    .createTextOutput(JSON.stringify({status:'OK', id:id}))
    .setMimeType(ContentService.MimeType.JSON);
}

// Lee la pestaña "Trazabilidad OP" (órdenes de producción) para autollenar
// el formulario de PT manual y el de Devoluciones sin tener que escribir
// todo a mano. Busca las columnas por su encabezado (no por posición fija).
// Columnas reales confirmadas en la hoja: OP, Fecha, Codigo, Producto,
// Lote, ..., Cantidad a Producir, ..., Materiales, ...
function jsonTrazabilidadOP_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName('Trazabilidad OP');
  const rows = [];
  if (sh && sh.getLastRow() > 1) {
    const vals = sh.getDataRange().getDisplayValues();
    const header = vals[0].map(function(h) { return String(h || '').trim().toLowerCase(); });
    const idxOp = header.findIndex(function(h) { return h === 'op'; });
    const idxCodigo = header.findIndex(function(h) {
      return h === 'codigo' || h === 'código' || h.indexOf('clave') !== -1 || h.indexOf('sku') !== -1;
    });
    const idxProducto = header.findIndex(function(h) { return h.indexOf('producto') !== -1; });
    const idxLote  = header.findIndex(function(h) { return h.indexOf('lote') !== -1; });
    const idxCant  = header.findIndex(function(h) { return h.indexOf('cant') !== -1 || h.indexOf('piezas') !== -1 || h.indexOf('unidades') !== -1; });
    const idxFecha = header.findIndex(function(h) { return h.indexOf('fecha') !== -1; });
    // Ojo: hay una columna "Daño a Materiales" que también contiene la
    // palabra "material" y aparece ANTES que la columna real "Materiales"
    // -- por eso aquí se busca coincidencia EXACTA, no por substring.
    const idxMateriales = header.findIndex(function(h) { return h === 'materiales'; });
    for (let i = 1; i < vals.length; i++) {
      const r = vals[i];
      const codigo = idxCodigo >= 0 ? String(r[idxCodigo] || '').trim() : '';
      if (!codigo) continue;
      rows.push({
        op: idxOp >= 0 ? String(r[idxOp] || '').trim() : '',
        clave: codigo, // alias -- lo sigue usando el modal de "Agregar PT a mano"
        codigo: codigo,
        producto: idxProducto >= 0 ? String(r[idxProducto] || '').trim() : '',
        lote: idxLote >= 0 ? String(r[idxLote] || '').trim() : '',
        cantidad: idxCant >= 0 ? String(r[idxCant] || '').trim() : '',
        fecha: idxFecha >= 0 ? String(r[idxFecha] || '').trim() : '',
        materiales: idxMateriales >= 0 ? String(r[idxMateriales] || '').trim() : ''
      });
    }
  }
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Devoluciones / Destrucciones =====
// Cada folio (papeleta) se guarda como VARIAS filas en la hoja
// "Devoluciones" -- una fila por renglón de material devuelto/destruido --
// todas compartiendo el mismo Id de folio, para poder agruparlas de vuelta
// al leerlas y para poder borrar el folio completo de un jalón si hace
// falta. El Tipo ('no_utilizado' o 'destruccion') se repite en cada fila.
function doPostDevolucion(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sh = ss.getSheetByName('Devoluciones');
  if (!sh) {
    sh = ss.insertSheet('Devoluciones');
    sh.appendRow(['Id','Folio','Tipo','OP','CodigoPT','ProductoPT','LotePT','FechaCaptura','Codigo','Descripcion','Lote','Cantidad','Registrado','NumMovimiento']);
  }

  const folio = String(data.folio || '').trim();
  const lineas = Array.isArray(data.lineas) ? data.lineas : [];
  if (!folio) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'Falta el folio.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (!lineas.length) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'El folio no tiene renglones de material.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const id = Utilities.getUuid();
  const tipo = String(data.tipo || '').trim();
  const op = String(data.op || '').trim();
  const codigoPT = String(data.codigoPT || '').trim();
  const productoPT = String(data.productoPT || '').trim();
  const lotePT = String(data.lotePT || '').trim();
  const fechaCaptura = String(data.fecha || '').trim();
  const registrado = String(data.registrado || '').trim();
  // Número de movimiento de SAI para esta papeleta (folio) -- se captura a
  // mano en el formulario, es el mismo para todos los renglones del
  // folio, y sirve para cruzarlo contra la columna "Movimiento" de MOVREP
  // (ver jsonMovRepDev_) y así saber si ya se puede hacer el traspaso de
  // Devoluciones a EYEJAL.
  const numMovimiento = String(data.numMovimiento || '').trim();

  const filaIni = sh.getLastRow() + 1;
  const out = lineas.map(function(l) {
    return [id, folio, tipo, op, codigoPT, productoPT, lotePT, fechaCaptura,
      String(l.codigo || '').trim(), String(l.descripcion || '').trim(),
      String(l.lote || '').trim(), l.cantidad || 0, registrado, numMovimiento];
  });
  const rango = sh.getRange(filaIni, 1, out.length, 14);
  rango.setNumberFormat('@');
  rango.setValues(out);

  return ContentService
    .createTextOutput(JSON.stringify({ status: 'OK', id: id }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPostEliminarDevolucion(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName('Devoluciones');
  if (sh) {
    const vals = sh.getDataRange().getValues();
    // De abajo hacia arriba para que borrar una fila no recorra los índices
    // de las que faltan por revisar.
    for (let i = vals.length - 1; i >= 1; i--) {
      if (String(vals[i][0]).trim() === String(data.id).trim()) {
        sh.deleteRow(i + 1);
      }
    }
  }
  return ContentService
    .createTextOutput(JSON.stringify({status:'OK'}))
    .setMimeType(ContentService.MimeType.JSON);
}

// Números de movimiento reales que trae MOVREP para el almacén DEV (los
// que vienen de movdev.XLS) -- se usa para cruzar contra el campo
// "Número de movimiento" que se captura a mano en cada folio de
// Devoluciones. Se manda deduplicado por movimiento (MOVREP trae una fila
// por cada clave dentro del mismo movimiento, no una fila por movimiento).
function jsonMovRepDev_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName('MOVREP');
  const porMov = {};
  if (sh && sh.getLastRow() > 1) {
    const vals = sh.getRange(2, 1, sh.getLastRow() - 1, 11).getDisplayValues();
    vals.forEach(function(r) {
      const almacen = String(r[2] || '').trim().toUpperCase();
      if (almacen !== 'DEV') return;
      const movimiento = String(r[0] || '').trim();
      if (!movimiento) return;
      if (!porMov[movimiento]) {
        porMov[movimiento] = {
          movimiento: movimiento,
          fecha: String(r[1] || '').trim(),
          tipo: String(r[3] || '').trim(),
          detalle: String(r[4] || '').trim(),
          lineas: 0,
          lineasDetalle: []
        };
      }
      porMov[movimiento].lineas++;
      // Clave (índice 6), Lote (índice 8) y Cantidad (índice 9) -- ver
      // HEADER_MOVREP en "actualizar inventario.txt". Se manda el detalle
      // real de cada línea para poder cruzarlo contra lo capturado en el
      // folio de Devoluciones (código + lote + cantidad).
      porMov[movimiento].lineasDetalle.push({
        clave: String(r[6] || '').trim(),
        lote: String(r[8] || '').trim(),
        cantidad: parseFloat(String(r[9] || '0').replace(/,/g, '')) || 0
      });
    });
  }
  return ContentService
    .createTextOutput(JSON.stringify(Object.values(porMov)))
    .setMimeType(ContentService.MimeType.JSON);
}

// Registró/Foto/Observaciones/Estado/Ubic...), de antes de que se
// agregaran los campos de OP/CodigoPT/ProductoPT/LotePT -- doPostDevolucion()
// nunca lo vuelve a escribir porque solo pone el encabezado la primera vez
// que crea la pestaña (si ya existe, la deja como está). LOS DATOS están
// bien, nomás los títulos de arriba no. Correr esta función UNA vez
// (desde script.google.com: abre este proyecto, elige
// corregirEncabezadoDevoluciones en el menú de arriba junto a "Ejecutar",
// dale Ejecutar) para reescribir nada más el renglón 1 con los títulos
// correctos -- no toca ninguna fila de datos.
function corregirEncabezadoDevoluciones() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName('Devoluciones');
  if (!sh) {
    Logger.log('No existe la pestaña "Devoluciones" todavía -- nada que corregir.');
    return;
  }

  sh.getRange(1, 1, 1, header.length).setValues([header]);

}

// ---- Plan de Embarque (correo semanal de Logística con el PDF del plan) ----
// Cada folio (GL#####) se guarda como varias filas -- una por renglón de
// material -- con los datos de cabecera (folio/fecha/transportista/etc.)
// repetidos en cada fila, igual que Devoluciones. El folio es único: si ya
// existe una fila con ese folio no se vuelve a insertar (evita duplicados
// cuando la tarea automática revisa el correo varias veces al día).
function doPostPlanEmbarque(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sh = ss.getSheetByName('PlanEmbarque');
  if (!sh) {
    sh = ss.insertSheet('PlanEmbarque');
    sh.appendRow(['Folio', 'Fecha', 'SolicitadoPor', 'Transportista', 'Retorno', 'Capacidad',
      'TotalPiezas', 'TotalTarimas', 'Renglon', 'Clave', 'Descripcion', 'Piezas', 'Lote',
      'Tarimas', 'AlmacenOrigen', 'FechaRegistro']);
  }

  const folio = String(data.folio || '').trim();
  const lineas = Array.isArray(data.lineas) ? data.lineas : [];
  if (!folio) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'Falta el folio del embarque.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (!lineas.length) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'El plan no tiene renglones de material.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Dedup por folio -- si ya está registrado no se vuelve a guardar.
  if (sh.getLastRow() > 1) {
    const folios = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getDisplayValues();
    for (let i = 0; i < folios.length; i++) {
      if (String(folios[i][0]).trim().toUpperCase() === folio.toUpperCase()) {
        return ContentService
          .createTextOutput(JSON.stringify({ status: 'SKIP', mensaje: 'El folio ' + folio + ' ya estaba registrado.' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
  }

  const fecha = String(data.fecha || '').trim();
  const solicitadoPor = String(data.solicitadoPor || '').trim();
  const transportista = String(data.transportista || '').trim();
  const retorno = String(data.retorno || '').trim();
  const capacidad = String(data.capacidad || '').trim();
  const totalPiezas = data.totalPiezas || 0;
  const totalTarimas = data.totalTarimas || 0;
  const fechaRegistro = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');

  const filaIni = sh.getLastRow() + 1;
  const out = lineas.map(function (l) {
    return [folio, fecha, solicitadoPor, transportista, retorno, capacidad, totalPiezas, totalTarimas,
      String(l.renglon || '').trim(), String(l.clave || '').trim(), String(l.descripcion || '').trim(),
      l.piezas || 0, String(l.lote || '').trim(), l.tarimas || 0, String(l.almacenOrigen || '').trim(),
      fechaRegistro];
  });
  const rango = sh.getRange(filaIni, 1, out.length, 16);
  rango.setNumberFormat('@');
  rango.setValues(out);

  return ContentService
    .createTextOutput(JSON.stringify({ status: 'OK', folio: folio, renglones: out.length }))
    .setMimeType(ContentService.MimeType.JSON);
}

function jsonPlanEmbarque_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName('PlanEmbarque');
  const rows = [];
  if (sh && sh.getLastRow() > 1) {
    const vals = sh.getRange(2, 1, sh.getLastRow() - 1, 16).getDisplayValues();
    // Índice de existencias de InventarioPT/Inventario, para marcar por
    // línea si esa clave+lote SÍ está en el almacén que dice "Almacén
    // Origen" del PDF (ej. "40 PT" -> InventarioPT/40PT, "REPROCESO" ->
    // Inventario/REPROJAL). Se construye una sola vez por llamada, no por
    // línea, para no releer las hojas repetidamente.
    const indiceAlmacenes = construirIndiceAlmacenesPlanEmbarque_();
    vals.forEach(function (r) {
      const folio = String(r[0] || '').trim();
      if (!folio) return;
      const clave = String(r[9] || '').trim();
      const lote = String(r[12] || '').trim();
      const almacenOrigen = String(r[14] || '').trim();
      rows.push({
        folio: folio,
        fecha: String(r[1] || '').trim(),
        solicitadoPor: String(r[2] || '').trim(),
        transportista: String(r[3] || '').trim(),
        retorno: String(r[4] || '').trim(),
        capacidad: String(r[5] || '').trim(),
        totalPiezas: parseFloat(String(r[6]).replace(/,/g, '')) || 0,
        totalTarimas: parseFloat(String(r[7]).replace(/,/g, '')) || 0,
        renglon: String(r[8] || '').trim(),
        clave: clave,
        descripcion: String(r[10] || '').trim(),
        piezas: parseFloat(String(r[11]).replace(/,/g, '')) || 0,
        lote: lote,
        tarimas: parseFloat(String(r[13]).replace(/,/g, '')) || 0,
        almacenOrigen: almacenOrigen,
        fechaRegistro: String(r[15] || '').trim(),
        // true = se encontró esa clave+lote en el almacén correspondiente;
        // false = NO se encontró (posible alerta); null = el texto de
        // "Almacén Origen" no se reconoce, no se pudo verificar.
        enAlmacen: verificarEnAlmacenPlanEmbarque_(indiceAlmacenes, almacenOrigen, clave, lote)
      });
    });
  }
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

// ---- Verificación de Plan de Embarque contra InventarioPT/Inventario ----
// El texto "Almacén Origen" que trae el PDF (ej. "40 PT", "REPROCESO") no
// coincide letra por letra con el valor real de la columna "Almacén" en las
// hojas de inventario (ej. "40PT", "REPROJAL") -- confirmado revisando datos
// reales del Sheet. Este mapa traduce uno al otro. Si en algún Plan de
// Embarque futuro aparece un almacén de origen distinto a estos dos, se
// puede agregar aquí una línea más.
// REPROJAL vive en "InventarioPT" (se movió ahí desde "Inventario" en
// actualizar inventario.txt, junto con 40PT/PTJAL/PTFGUAD) -- no en
// "Inventario" (que ahora es solo EYEJAL/40EYE).
function mapAlmacenPlanEmbarque_(almacenOrigenTexto) {
  const t = String(almacenOrigenTexto || '').toUpperCase().replace(/\s+/g, '');
  if (t.indexOf('40PT') !== -1) return { hoja: 'InventarioPT', almacen: '40PT' };
  if (t.indexOf('40EYE') !== -1) return { hoja: 'Inventario', almacen: '40EYE' };
  if (t.indexOf('REPRO') !== -1) return { hoja: 'InventarioPT', almacen: 'REPROJAL' };
  return null;
}

// Lee InventarioPT e Inventario UNA sola vez y arma, por hoja, un set de
// llaves "almacen|clave|lote" para poder verificar cada línea del Plan de
// Embarque con una simple búsqueda en memoria (en vez de recorrer la hoja
// completa por cada línea, que sería mucho más lento).
function construirIndiceAlmacenesPlanEmbarque_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const indices = {};
  ['InventarioPT', 'Inventario'].forEach(function (nombreHoja) {
    const sh = ss.getSheetByName(nombreHoja);
    const set = {};
    if (sh && sh.getLastRow() > 1) {
      // Columnas 1-8: Clase, Clave, Descripción, Lote, Existencia, Unidad,
      // Fecha Caducidad, Almacén (igual en ambas hojas).
      const vals = sh.getRange(2, 1, sh.getLastRow() - 1, 8).getDisplayValues();
      vals.forEach(function (r) {
        const clave = String(r[1] || '').trim().toUpperCase();
        const lote = String(r[3] || '').trim().toUpperCase();
        const almacen = String(r[7] || '').trim().toUpperCase();
        if (!clave || !lote || !almacen) return;
        set[almacen + '|' + clave + '|' + lote] = true;
      });
    }
    indices[nombreHoja] = set;
  });
  return indices;
}

function verificarEnAlmacenPlanEmbarque_(indices, almacenOrigenTexto, clave, lote) {
  const mapa = mapAlmacenPlanEmbarque_(almacenOrigenTexto);
  if (!mapa) return null;
  const set = indices[mapa.hoja] || {};
  const key = mapa.almacen + '|' + String(clave || '').trim().toUpperCase() + '|' + String(lote || '').trim().toUpperCase();
  return !!set[key];
}

// ---- Plan de Embarque: lectura automática desde Drive (Power Automate) ----
// Reemplaza al correo->HTTP directo (bloqueado por licencia Premium de Power
// Automate): ahora el flujo de Power Automate solo GUARDA el PDF adjunto del
// correo "Plan de embarque" en la carpeta de Drive "PlanEmbarque_PDFs" (de la
// MISMA cuenta de Google dueña de este proyecto/Sheet). Esta función, en un
// disparador de tiempo (cada hora por ejemplo), revisa esa carpeta, convierte
// cada PDF a texto vía OCR, extrae los mismos campos que ya pedía
// doPostPlanEmbarque() y llama a esa función directamente (mismo proyecto,
// sin necesidad de HTTP). El PDF ya procesado se mueve a
// "PlanEmbarque_PDFs_Procesados" para no volver a leerlo. Si un PDF no se
// pudo interpretar (folio no encontrado, etc.) se deja SIN mover en la
// carpeta de entrada para revisión manual, y queda registrado en Logger.
//
// REQUIERE: en el editor de Apps Script > Servicios (ícono +) > agregar el
// servicio avanzado "Drive API" (Drive, versión 2 o 3) -- sin esto,
// Drive.Files.insert/copy no existen y truena con "Drive is not defined".
//
// Para activar el disparador automático: correr UNA vez la función
// instalarTriggerPlanEmbarque() desde este editor (selecciónala en el menú de
// arriba, junto a "Ejecutar"). Con eso queda programada para correr sola cada
// hora sin que dependa de que la laptop esté encendida.
function revisarPlanEmbarquePDFs() {
  const carpeta = obtenerOCrearCarpetaDrive_('PlanEmbarque_PDFs');
  const carpetaProcesados = obtenerOCrearCarpetaDrive_('PlanEmbarque_PDFs_Procesados');
  const archivos = carpeta.getFilesByType(MimeType.PDF);

  while (archivos.hasNext()) {
    const archivo = archivos.next();
    const nombre = archivo.getName();
    try {
      const texto = extraerTextoPdfOcr_(archivo);
      const datos = parsearPlanEmbarquePdf_(texto);

      if (!datos || !datos.folio || !datos.lineas.length) {
        Logger.log('No se pudo interpretar "' + nombre + '" (folio o renglones no encontrados) -- se deja en la carpeta para revisión manual.');
        continue;
      }

      const resp = doPostPlanEmbarque(datos);
      Logger.log('Procesado "' + nombre + '": ' + resp.getContent());

      // Se mueve al archivo de procesados tanto si se guardó (OK) como si ya
      // existía (SKIP) -- en ambos casos ya no hay nada más que hacer con
      // ese PDF. Si hubo un error real, doPostPlanEmbarque lanza una
      // excepción capturada más abajo y el PDF se queda sin mover.
      archivo.moveTo(carpetaProcesados);
    } catch (err) {
      Logger.log('Error procesando "' + nombre + '": ' + err);
    }
  }
}

function instalarTriggerPlanEmbarque() {
  // Quita cualquier disparador viejo de esta misma función antes de crear
  // uno nuevo, para poder correr esta instaladora otra vez sin duplicar.
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'revisarPlanEmbarquePDFs') {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger('revisarPlanEmbarquePDFs')
    .timeBased()
    .everyHours(1)
    .create();
  Logger.log('Listo: revisarPlanEmbarquePDFs() ahora corre sola cada hora.');
}

function obtenerOCrearCarpetaDrive_(nombre) {
  const it = DriveApp.getFoldersByName(nombre);
  if (it.hasNext()) return it.next();
  return DriveApp.createFolder(nombre);
}

// Convierte el PDF a un Google Doc temporal CON OCR (servicio avanzado de
// Drive), lee su texto y borra el Doc temporal (el PDF original no se toca
// aquí -- eso lo hace revisarPlanEmbarquePDFs() al moverlo a "Procesados").
function extraerTextoPdfOcr_(archivoPdf) {
  // "convert: true" es el que le dice a Drive que convierta el PDF a Google
  // Doc -- sin este parámetro, Drive sube el archivo tal cual (como PDF) y
  // truena al pedirle OCR ("OCR is not supported for files of type
  // application/vnd.google-apps.document"), porque nunca llegó a convertirlo.
  const recurso = {
    title: 'OCR_temp_' + archivoPdf.getId()
  };
  const archivoOcr = Drive.Files.insert(recurso, archivoPdf.getBlob(), { convert: true, ocr: true, ocrLanguage: 'es' });
  try {
    const doc = DocumentApp.openById(archivoOcr.id);
    return doc.getBody().getText();
  } finally {
    DriveApp.getFileById(archivoOcr.id).setTrashed(true);
  }
}

// Extrae del texto OCR los mismos campos que ya esperaba doPostPlanEmbarque():
// folio, fecha, solicitadoPor, transportista, retorno, capacidad,
// totalPiezas, totalTarimas y las líneas de la tabla (renglon, clave,
// descripcion, piezas, lote, tarimas, almacenOrigen).
//
// IMPORTANTE (verificado con el texto OCR real de GL10383): la conversión de
// Drive NO mete un salto de línea por cada renglón de la tabla -- toda la
// tabla sale como UNA sola línea larga, tipo:
//   "A PT-TOR-004 TORONGIA GEL CORP REAF 475G 720 26060387 1.0 REPROCESO
//    B/C/D PT-TOR-004 TORONGIA GEL CORP REAF 475G 2,880 26070439 3.0
//    REPROCESO E MAQ-MEDI-011 ..."
// Por eso aquí NO se separa por líneas para la tabla: se tokeniza por
// espacios y se recorre a mano, detectando dónde empieza cada renglón por su
// FORMA (letra de renglón + clave), no por saltos de línea. La descripción a
// veces trae números pegados a letras (ej. "475G", "125 ML") que podrían
// confundirse con la columna de piezas -- para evitarlo, un token solo se
// acepta como "piezas" si el token que sigue tiene forma de lote (varios
// dígitos) Y el que sigue a ese tiene forma de tarimas (decimal).
function parsearPlanEmbarquePdf_(texto) {
  const lineasTexto = texto.split(/\r?\n/).map(function (l) { return l.trim(); }).filter(Boolean);

  // ---- Encabezado: también viene todo en una sola línea, ej.:
  // "EMBARQUE # GL10383 FECHA 15-Sep-2026 SOLICITADO POR J.BALLESTEROS
  //  TRANSPORTISTA ALAVAMA RETORNO CAPACIDAD 14"
  // RETORNO puede venir vacío (sin valor) si no aplica, como en este caso.
  const lineaEncabezado = lineasTexto.find(function (l) {
    return /EMBARQUE\s*#?\s*[A-Z0-9]/i.test(l) && /CAPACIDAD/i.test(l);
  }) || '';

  let folio = '', fecha = '', solicitadoPor = '', transportista = '', retorno = '', capacidad = '';
  const mEnc = lineaEncabezado.match(
    /EMBARQUE\s*#?\s*([A-Z0-9]+)\s+FECHA\s+(\S+)\s+SOLICITADO\s*POR\s+(.*?)\s+TRANSPORTISTA\s+(.*?)\s+RETORNO\s*(.*?)\s*CAPACIDAD\s+(\S+)/i
  );
  if (mEnc) {
    folio = mEnc[1].trim();
    fecha = mEnc[2].trim();
    solicitadoPor = mEnc[3].trim();
    transportista = mEnc[4].trim();
    retorno = mEnc[5].trim();
    capacidad = mEnc[6].trim();
  }

  // ---- Tabla: se busca la línea que empieza con el encabezado de columnas
  // ("# CLAVE DESCRIPCIÓN PIEZAS LOTE TARIMAS ORIGEN ALMACÉN ...") y se le
  // quita ese encabezado, dejando solo los datos para tokenizar.
  const lineaTabla = lineasTexto.find(function (l) { return /^#\s*CLAVE/i.test(l); }) || '';
  const tokens = lineaTabla
    .replace(/^#\s*CLAVE\s*DESCRIPCI[ÓO]N\s*PIEZAS\s*LOTE\s*TARIMAS\s*ORIGEN\s*ALMAC[ÉE]N/i, '')
    .trim()
    .split(/\s+/)
    .filter(Boolean);

  const esRenglon = function (t) { return /^[A-Z](?:\/[A-Z])*$/.test(t || ''); };
  const esClave = function (t) { return /^[A-Z]{2,6}(?:-[A-Z0-9]+){1,3}$/.test(t || ''); };
  const pareceLote = function (t) { return /^\d{5,10}(?:\(\d{4,10}\))?$/.test(t || ''); };
  const esTarimas = function (t) { return /^\d+(?:\.\d+)?$/.test(t || ''); };
  // Un token solo cuenta como "piezas" si los DOS tokens siguientes tienen
  // forma de lote y de tarimas -- así no se confunde con números sueltos
  // dentro de la descripción (ej. "475G" no matchea, pero "125" en "125 ML"
  // sí tiene forma de número; el chequeo de los siguientes tokens lo filtra).
  const esPiezas = function (arr, idx) {
    return /^[\d,]+$/.test(arr[idx] || '') && pareceLote(arr[idx + 1]) && esTarimas(arr[idx + 2]);
  };

  const lineas = [];
  let i = 0;
  while (i < tokens.length) {
    if (!esRenglon(tokens[i]) || !esClave(tokens[i + 1])) { i++; continue; }
    const renglon = tokens[i]; i++;
    const clave = tokens[i]; i++;

    const descPartes = [];
    while (i < tokens.length && !esPiezas(tokens, i)) { descPartes.push(tokens[i]); i++; }
    if (i >= tokens.length) break; // fila incompleta (no se encontró piezas) -- se descarta

    const piezas = parseFloat(tokens[i].replace(/,/g, '')) || 0; i++;
    const lote = tokens[i] || ''; i++;
    const tarimas = parseFloat(tokens[i]) || 0; i++;

    const almacenPartes = [];
    while (i < tokens.length && !(esRenglon(tokens[i]) && esClave(tokens[i + 1]))) {
      almacenPartes.push(tokens[i]); i++;
    }

    lineas.push({
      renglon: renglon,
      clave: clave,
      descripcion: descPartes.join(' '),
      piezas: piezas,
      lote: lote,
      tarimas: tarimas,
      almacenOrigen: almacenPartes.join(' ')
    });
  }

  // ---- Totales: aparecen en su propia línea corta, ej. "27,818 14.0",
  // después de la línea que dice "TOTALES".
  let totalPiezas = 0, totalTarimas = 0;
  const idxTotales = lineasTexto.findIndex(function (l) { return /^TOTALES$/i.test(l); });
  if (idxTotales >= 0) {
    for (let k = idxTotales + 1; k < lineasTexto.length; k++) {
      const mTot = lineasTexto[k].match(/^([\d,]+)\s+(\d+(?:\.\d+)?)$/);
      if (mTot) {
        totalPiezas = parseFloat(mTot[1].replace(/,/g, '')) || 0;
        totalTarimas = parseFloat(mTot[2]) || 0;
        break;
      }
    }
  }
  // Si no se encontró la fila de totales, se calculan sumando las líneas --
  // mejor un total aproximado que dejarlo en 0.
  if (!totalPiezas) totalPiezas = lineas.reduce(function (a, l) { return a + l.piezas; }, 0);
  if (!totalTarimas) totalTarimas = lineas.reduce(function (a, l) { return a + l.tarimas; }, 0);

  return {
    folio: folio,
    fecha: fecha,
    solicitadoPor: solicitadoPor,
    transportista: transportista,
    retorno: retorno,
    capacidad: capacidad,
    totalPiezas: totalPiezas,
    totalTarimas: totalTarimas,
    lineas: lineas
  };
}

function doPostConteo(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);

  let hist = ss.getSheetByName('Historial');
  if(!hist) {
    hist = ss.insertSheet('Historial');
    hist.appendRow(['Fecha','Clave','Lote','Cantidad','Registrado','Causa']);
  }
  hist.appendRow([data.fechaConteo, data.clave, data.lote, data.cantidad, data.registrado, data.causa||'']);

  const cic = ss.getSheetByName('Ciclicos');
  const vals = cic.getDataRange().getValues();
  for(let i = 1; i < vals.length; i++) {
    if(vals[i][1] == data.clave && vals[i][3] == data.lote) {
      cic.getRange(i+1, 9).setValue(data.fechaConteo);
      cic.getRange(i+1, 10).setValue(data.cantidad);
      cic.getRange(i+1, 11).setValue(data.registrado);
      break;
    }
  }

  // Si el código se contó estando programado para otra semana (atrasado o
  // adelantado), el frontend manda semanaDestino/semanaFechaDestino: se
  // reasigna TODA la clave (todos sus lotes, no solo el que se acaba de
  // contar) a la semana en curso, para que no quede desfasada entre lotes.
  if (data.semanaDestino) {
    for (let i = 1; i < vals.length; i++) {
      if (vals[i][1] == data.clave) {
        cic.getRange(i + 1, 13).setValue(data.semanaDestino);
        cic.getRange(i + 1, 14).setValue(data.semanaFechaDestino || '');
      }
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({status:'OK'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  const accion = e.parameter && e.parameter.accion;
  // Lectura en vivo para Historial/Ajustes: "Publicar en la web" (el CSV que
  // usaba antes el HTML) tiene su propio caché de varios minutos del lado
  // de Google que no se puede forzar con "&t=...". Leyendo directo del
  // Sheet aquí, la app ve los conteos/ajustes al instante después de
  // guardarlos.
  if (accion === 'historial') return jsonHistorial_();
  if (accion === 'ajustes') return jsonAjustes_();
  if (accion === 'ubicaciones') return jsonUbicaciones_();
  if (accion === 'catalogo_ubicaciones') return jsonCatalogoUbicaciones_();
  if (accion === 'pt_manual') return jsonPtManual_();
  if (accion === 'trazabilidad_op') return jsonTrazabilidadOP_();
  if (accion === 'atrasados_excluidos') return jsonAtrasadosExcluidos_();
  if (accion === 'devoluciones') return jsonDevoluciones_();
  if (accion === 'acond') return jsonAcond_();
  if (accion === 'movrep_dev') return jsonMovRepDev_();
  if (accion === 'plan_embarque') return jsonPlanEmbarque_();
  // Fallback por si algo no puede hacer un POST real (ej. una tarea
  // automatizada limitada a GET): manda el mismo payload de
  // doPostPlanEmbarque pero como JSON codificado en el parámetro ?payload=.
  if (accion === 'registrar_plan_embarque') {
    try {
      return doPostPlanEmbarque(JSON.parse(e.parameter.payload));
    } catch (err) {
      return ContentService
        .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: String(err) }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  return ContentService
    .createTextOutput(JSON.stringify({status:'OK'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function jsonUbicaciones_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const ubic = ss.getSheetByName('Ubicaciones');
  const rows = [];
  if (ubic && ubic.getLastRow() > 1) {
    // 8 columnas (Tipo/Motivo se agregaron para soportar ubicaciones
    // bloqueadas -- ver doPostUbicacion). Filas viejas de antes de ese
    // cambio no tienen columnas 7-8: salen vacías, no truena.
    const vals = ubic.getRange(2, 1, ubic.getLastRow() - 1, 8).getDisplayValues();
    vals.forEach(function(r) {
      const clave = String(r[1] || '').trim();
      const lote = String(r[2] || '').trim();
      const tipo = String(r[6] || '').trim();

      // ocupada por otro departamento, no un lote tuyo) -- se incluye de
      // todas formas. Cualquier otra fila sin clave/lote es basura/vacía y
      // se sigue descartando, igual que antes.
      if (!clave || !lote) { if (tipo !== 'Bloqueada') return; }
      rows.push({
        id: String(r[0] || '').trim(),
        clave: clave,
        lote: lote,
        ubicacion: String(r[3] || '').trim(),
        registrado: String(r[4] || '').trim(),
        fecha: String(r[5] || '').trim(),
        tipo: tipo,
        motivo: String(r[7] || '').trim()
      });
    });
  }
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

// guardarUbicacion() en el HTML). No duplica: si el código ya existe
// (sin importar mayúsculas/minúsculas) regresa OK sin agregar otra fila.
function doPostAgregarUbicacionCatalogo(data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sh = ss.getSheetByName('CatalogoUbicaciones');
  if (!sh) {
    sh = ss.insertSheet('CatalogoUbicaciones');
    sh.appendRow(['Ubicacion', 'Registrado', 'Fecha']);
  }
  const ubicacion = String(data.ubicacion || '').trim();
  if (!ubicacion) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ERROR', mensaje: 'Falta la ubicación.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (sh.getLastRow() > 1) {
    const existentes = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getDisplayValues();
    for (let i = 0; i < existentes.length; i++) {
      if (String(existentes[i][0] || '').trim().toLowerCase() === ubicacion.toLowerCase()) {
        return ContentService
          .createTextOutput(JSON.stringify({ status: 'OK', mensaje: 'Ya existía en el catálogo.' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
  }
  const fila = sh.getLastRow() + 1;
  const rango = sh.getRange(fila, 1, 1, 3);
  rango.setNumberFormat('@'); // mismo blindaje que Ubicaciones: evita notación científica
  rango.setValues([[ubicacion, data.registrado || '', data.fecha || '']]);
  return ContentService
    .createTextOutput(JSON.stringify({ status: 'OK' }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Importación masiva de Ubicaciones desde Excel =====
// Para no tener que meter cada ubicación una por una desde el HTML (que le
// pone un Id nuevo cada vez), esta función hace lo mismo pero para muchas
// filas de un jalón:
//   1. En este mismo Sheet crea (o usa) la pestaña "Importar Ubicaciones"
//      con columnas: Clave | Lote | Ubicacion | Registrado | Fecha
//      (Registrado y Fecha son opcionales, se pueden dejar en blanco).
//   2. Pega ahí los datos de tu Excel (en ese orden de columnas).
//      IMPORTANTE: antes de pegar, selecciona la columna "Ubicacion" y
//      ponle Formato > Número > Texto sin formato -- si no, Sheets puede
//      convertir ubicaciones como "07E03" en el número 7000.
//   3. Abre este proyecto en script.google.com, elige la función
//      importarUbicacionesMasivo en el menú de arriba (junto al botón
//      "Ejecutar") y dale Ejecutar.
//   4. Revisa el resultado en Ver > Registros (o Ctrl+Enter) -- ahí dice
//      cuántas se agregaron. Ya quedan disponibles en la app al instante.
// Se puede correr varias veces sin duplicar nada, incluso si vuelves a
// pegar el Excel completo desde cero (ej. porque ya no te acuerdas cuáles
// habías subido): cada fila que ya se importó queda marcada en la columna
// "Importado" y se salta la próxima vez, Y ADEMÁS cualquier fila cuya
// Clave+Lote+Ubicación ya exista en la pestaña "Ubicaciones" (aunque no
// venga de una importación anterior, por ejemplo si se capturó a mano desde
// la app) se salta también, marcada como "YA EXISTÍA".
function importarUbicacionesMasivo() {
  const ss = SpreadsheetApp.openById(SHEET_ID);

  let stage = ss.getSheetByName('Importar Ubicaciones');
  if (!stage) {
    stage = ss.insertSheet('Importar Ubicaciones');
    stage.appendRow(['Clave', 'Lote', 'Ubicacion', 'Registrado', 'Fecha', 'Importado']);
    Logger.log('Se creó la pestaña "Importar Ubicaciones". Pega ahí tus datos (Clave, Lote, Ubicacion) y vuelve a correr esta función.');
    return;
  }

  let ubic = ss.getSheetByName('Ubicaciones');
  if (!ubic) {
    ubic = ss.insertSheet('Ubicaciones');
    ubic.appendRow(['Id', 'Clave', 'Lote', 'Ubicacion', 'Registrado', 'Fecha']);
  }

  const lastRow = stage.getLastRow();
  if (lastRow < 2) {
    Logger.log('La pestaña "Importar Ubicaciones" no tiene filas todavía.');
    return;
  }

  // Para poder volver a subir el Excel completo (ej. cuando ya no te
  // acuerdas cuáles habías puesto) sin duplicar nada, se arma un set con
  // "clave|lote|ubicacion" de TODO lo que ya existe en la pestaña
  // "Ubicaciones" (sin importar si llegó por importación o capturado a mano
  // desde la app), y se salta cualquier fila del Excel que ya coincida.
  const yaExiste = {};
  if (ubic.getLastRow() > 1) {
    const existentes = ubic.getRange(2, 1, ubic.getLastRow() - 1, 4).getValues();
    existentes.forEach(function (r) {
      const k = String(r[1] || '').trim().toLowerCase() + '|' + String(r[2] || '').trim().toLowerCase() + '|' + String(r[3] || '').trim().toLowerCase();
      yaExiste[k] = true;
    });
  }

  const vals = stage.getRange(2, 1, lastRow - 1, 6).getValues();
  const hoy = Utilities.formatDate(new Date(), 'America/Mexico_City', 'dd/MM/yyyy HH:mm:ss');
  let agregadas = 0;
  let saltadas = 0;
  let repetidas = 0;

  for (let i = 0; i < vals.length; i++) {
    const clave = vals[i][0], lote = vals[i][1], ubicacion = vals[i][2], registrado = vals[i][3], fecha = vals[i][4], importado = vals[i][5];
    if (importado) continue; // ya se importó antes, se salta

    const claveStr = String(clave || '').trim();
    const loteStr = String(lote || '').trim();
    const ubicStr = String(ubicacion || '').trim();
    if (!claveStr || !loteStr) { saltadas++; continue; } // fila vacía o sin clave/lote

    const key = claveStr.toLowerCase() + '|' + loteStr.toLowerCase() + '|' + ubicStr.toLowerCase();
    if (yaExiste[key]) {
      // Esa clave+lote+ubicación ya está en "Ubicaciones" -- no se duplica,
      // pero SÍ se marca como importada para que no se vuelva a revisar cada
      // vez que corras la función.
      stage.getRange(i + 2, 6).setValue('YA EXISTÍA · ' + hoy);
      repetidas++;
      continue;
    }

    const id = Utilities.getUuid();
    const filaDestino = ubic.getLastRow() + 1;
    const rango = ubic.getRange(filaDestino, 1, 1, 6);
    // Mismo blindaje que doPostUbicacion: fuerza texto plano ANTES de
    // escribir para que una ubicación como "07E03" no se guarde como número.
    rango.setNumberFormat('@');
    rango.setValues([[id, claveStr, loteStr, ubicStr, String(registrado || 'Importado').trim(), String(fecha || hoy).trim()]]);

    stage.getRange(i + 2, 6).setValue('SI · ' + hoy);
    yaExiste[key] = true; // por si el Excel trae la misma fila repetida más de una vez
    agregadas++;
  }

  Logger.log('Listo: ' + agregadas + ' ubicaciones nuevas agregadas a "Ubicaciones". ' + repetidas + ' ya existían y se saltaron (no se duplicaron). ' + saltadas + ' filas se saltaron por no tener Clave y Lote.');
}

function doPostAjuste(e, data) {
  const ss = SpreadsheetApp.openById(SHEET_ID);

  let ajustes = ss.getSheetByName('Ajustes');
  if(!ajustes) {
    ajustes = ss.insertSheet('Ajustes');
    ajustes.appendRow(['Fecha Ajuste','Clave','Lote','Cantidad Contada','Cantidad SAI Anterior','Registrado','Causa']);
  }
  // Por si la hoja ya existía sin la columna Causa, la agrega sola
  if (String(ajustes.getRange(1,7).getValue() || '').trim() !== 'Causa') {
    ajustes.getRange(1,7).setValue('Causa');
  }

  ajustes.appendRow([
    data.fechaAjuste,
    data.clave,
    data.lote,
    data.cantidadContada,
    data.cantidadSAI,
    data.registrado,
    data.causa || ''
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({status:'OK'}))
    .setMimeType(ContentService.MimeType.JSON);
}
